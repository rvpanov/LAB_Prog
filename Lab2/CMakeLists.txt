cmake_minimum_required(VERSION 3.16)
project(Lab2SVGVariant1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(svg_generator src/main.cpp src/SVG.cpp)
target_include_directories(svg_generator PRIVATE include)
if(MSVC)
    target_compile_options(svg_generator PRIVATE /W4)
else()
    target_compile_options(svg_generator PRIVATE -Wall -Wextra -Wpedantic)
endif()
